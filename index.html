function doPost(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const data = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 1. LOGIKA PENYIMPANAN DATA (Tahap 1) if (data.action === 'save_data') { // Cek rate limit sederhana berdasarkan WA const wa = data.wa; const rows = sheet.getDataRange().getValues(); let count = 0; for (let i = 1; i < rows.length; i++) { if (rows[i][4] == wa && rows[i][0] == data.nama) { // Asumsi kolom 0=Nama, 4=WA count++; } } if (count >= 2) { return ContentService.createTextOutput(JSON.stringify({ status: 'rate_limited' })).setMimeType(ContentService.MimeType.JSON); } // Simpan ke Sheet (Tambahkan kolom "Status" di paling kanan jika belum ada) // Urutan kolom: Timestamp, Nama, Tgl Lahir, Gender, WA, Hasil Survey, Kegagalan, Bisikan, Penyerahan, Ref, Status sheet.appendRow([ new Date(), data.nama, data.tgl_lahir, data.gender, data.wa, data.hasil_survey, data.jawaban_kegagalan || '', data.jawaban_bisikan || '', data.jawaban_penyerahan || '', data.ref || 'DIRECT', 'Menunggu Klik WA' // <--- STATUS PENTING ]); return ContentService.createTextOutput(JSON.stringify({ status: 'saved' })).setMimeType(ContentService.MimeType.JSON); } // 2. LOGIKA PENGIRIMAN WHATSAPP (Tahap 2 - Dipicu saat user klik tombol WA) if (data.action === 'trigger_send_report') { const targetWa = data.wa; const targetNama = data.nama; // Cari baris data user ini untuk update status dan ambil data report const rows = sheet.getDataRange().getValues(); let rowIndex = -1; let userData = {}; for (let i = rows.length - 1; i >= 1; i--) { if (rows[i][4] == targetWa && rows[i][1] == targetNama) { rowIndex = i + 1; // 1-based index userData = { nama: rows[i][1], hasil: rows[i][5], status: rows[i][10] }; break; } } if (rowIndex === -1) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Data tidak ditemukan' })).setMimeType(ContentService.MimeType.JSON); } if (userData.status === 'Terkirim') { return ContentService.createTextOutput(JSON.stringify({ status: 'already_sent' })).setMimeType(ContentService.MimeType.JSON); } // === TEMPAT ANDA MENARUH API WHATSAPP ANDA === // Contoh menggunakan Fonnte/Wablas/Meta Cloud API. Ganti dengan endpoint Anda. // const apiUrl = 'https://api.fonnte.com/send'; // const token = 'TOKEN_API_ANDA'; // const message = `Halo ${userData.nama} 🕊️,\n\nTerima kasih telah menyelesaikan refleksi di prayer.my.id.\n\nBerikut adalah ringkasan Peta Hati Anda:\n${userData.hasil}\n\nJika ada yang ingin didiskusikan, kami siap mendengar. Tuhan menyertai Anda.`; /* const response = UrlFetchApp.fetch(apiUrl, { method: 'post', headers: { 'Authorization': token }, payload: { 'target': targetWa, 'message': message } }); */ // Update Status di Google Sheets menjadi 'Terkirim' sheet.getRange(rowIndex, 11).setValue('Terkirim'); // Kolom ke-11 adalah Status return ContentService.createTextOutput(JSON.stringify({ status: 'triggered' })).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } finally { lock.releaseLock(); } }
